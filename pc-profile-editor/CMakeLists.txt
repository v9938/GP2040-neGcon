cmake_minimum_required(VERSION 3.20)
project(PsxProfileEditor LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  add_compile_options(/W4 /utf-8 /permissive-)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
add_library(preset STATIC src/preset.cpp)
target_include_directories(preset PUBLIC src)
add_executable(preset_tests tests/preset_tests.cpp)
target_link_libraries(preset_tests PRIVATE preset)
enable_testing()
add_test(NAME preset_tests COMMAND preset_tests)
add_test(NAME shared_txt_sample COMMAND preset_tests --roundtrip "${CMAKE_CURRENT_SOURCE_DIR}/../preset-format/example.gp2040-psx.txt")
add_test(NAME legacy_txt_sample COMMAND preset_tests --roundtrip "${CMAKE_CURRENT_SOURCE_DIR}/../preset-format/legacy-v1.gp2040-psx.txt")
add_executable(xe1aj_tests tests/xe1aj_tests.cpp)
target_include_directories(xe1aj_tests PRIVATE ../headers)
target_compile_options(xe1aj_tests PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/UNDEBUG>)
add_test(NAME xe1aj_tests COMMAND xe1aj_tests)
add_executable(jogcon_tests tests/jogcon_tests.cpp ../src/jogcon.cpp)
target_include_directories(jogcon_tests PRIVATE ../headers)
target_compile_options(jogcon_tests PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/UNDEBUG>)
add_test(NAME jogcon_tests COMMAND jogcon_tests)
add_executable(negcon_additive_tests tests/negcon_additive_tests.cpp ../src/analog_preview.cpp)
target_include_directories(negcon_additive_tests PRIVATE ../headers)
target_compile_options(negcon_additive_tests PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/UNDEBUG> $<$<NOT:$<CXX_COMPILER_ID:MSVC>>:-UNDEBUG>)
add_test(NAME negcon_additive_tests COMMAND negcon_additive_tests)
if(WIN32)
  add_executable(psx_profile_editor WIN32 src/main.cpp src/files.cpp)
  target_compile_definitions(psx_profile_editor PRIVATE UNICODE _UNICODE NOMINMAX)
  target_link_libraries(psx_profile_editor PRIVATE preset user32 gdi32 comctl32 comdlg32 shell32)
  add_executable(preset_file_tests tests/file_tests.cpp src/files.cpp)
  target_compile_definitions(preset_file_tests PRIVATE UNICODE _UNICODE NOMINMAX)
  target_link_libraries(preset_file_tests PRIVATE preset)
  add_test(NAME preset_file_tests COMMAND preset_file_tests)
  add_test(NAME native_ui_smoke COMMAND psx_profile_editor --smoke-ui)
  set_tests_properties(native_ui_smoke PROPERTIES TIMEOUT 30)
  add_library(analog_preview SHARED ../src/analog_preview.cpp)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../src/jogcon.cpp")
    target_sources(analog_preview PRIVATE ../src/jogcon.cpp)
  endif()
  target_include_directories(analog_preview PRIVATE ../headers)
  target_compile_definitions(analog_preview PRIVATE ANALOG_PREVIEW_DLL NOMINMAX)
endif()
